# ── Core inference stack ──────────────────────────────────────────────────────
# The LoRA adapters under videos_checkpoints/ and images_checkpoints/ were
# exported by peft 0.19.x. Their adapter_config.json carries fields
# (arrow_config, alora_invocation_tokens, qalora_group_size, ...) that older
# peft releases reject, so do NOT pin peft back below 0.19 — the old
# >=0.10,<0.15 pin cannot load these checkpoints at all.
# torch>=2.1 alone is not a safe floor: 2.1.2 lacks torch.float8_e4m3fnuz,
# which transformers>=4.49 references unconditionally in its internals,
# producing "module 'torch' has no attribute 'float8_e4m3fnuz'" on every
# single model load regardless of which checkpoint. Observed reproducibly
# with torch 2.1.2 + transformers 4.49.0; multiple independent reports
# confirm the attribute exists from torch 2.3 onward. >=2.3 is the real
# floor for this transformers version, not >=2.1.
torch>=2.3
torchvision>=0.18
numpy>=1.26
transformers>=4.45
peft>=0.19
accelerate>=0.26.0
safetensors>=0.4.2
opencv-python>=4.8.0
pillow>=10.0.0

# MTCNN face detection: the video face-crop branch and the
# image_ffpp_facecrop checkpoint both need it. If it is absent the pipeline
# degrades gracefully — the face branch is skipped and the image ensemble
# re-weights itself onto the generalist checkpoints.
facenet-pytorch>=2.5.3

# ── API ───────────────────────────────────────────────────────────────────────

uvicorn[standard]
python-multipart

# ── Asynchronous task orchestration (M5) ──────────────────────────────────────
# Celery moves inference off the API process onto worker processes, so several
# uploads are analysed at once instead of serialising behind one event loop,
# and a failed run is retried rather than reported to the user as an error.
# Redis is both the broker and the hot-result cache (app/queue/cache.py).
#
# celery>=5.4 for Python 3.12+/3.13 support: 5.3.x pulls a billiard that fails
# to build there. redis>=5 because app/queue/state.py relies on the client
# exposing zrank/zadd through the same connection pool used for pub/sub.
celery>=5.4
redis>=5.0

# Server-side media download (M4 FE-1). httpx rather than requests because the
# streaming API makes it straightforward to abort a transfer mid-body once it
# passes the size cap, and because redirects can be disabled and followed
# manually — app/security/urlfetch.py re-validates the destination at every hop
# to stop a public URL redirecting onto an internal address.
httpx>=0.27

# ── Audio (M6, still a stub engine) ───────────────────────────────────────────
# Needed only once deeptruth_pipeline/inferencers/audio.py replaces
# AudioFakeNetStub; harmless to install ahead of that.
soundfile>=0.12.1
scipy>=1.11

# ── Forensic report export (M10 FE-3) ─────────────────────────────────────────
# Server-side PDF. ReportLab rather than a headless browser (which would mean
# shipping ~300MB of Chromium to render a page the server must also be able to
# reach) or a client-side canvas capture (which rasterises everything and loses
# selectable text — most of the point for a legal or journalistic document).
# Pure Python, no system libraries, unlike WeasyPrint's pango/cairo dependency.
reportlab>=4.0
